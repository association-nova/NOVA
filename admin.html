async function loadReq() { await fetchMembers(); const box = $('#reqList'); box.replaceChildren(); // دریافت درخواست‌ها از جدول Supabase const { data, error } = await supabaseClient .from('membership_requests') .select('id, name, cls, shad, email, status, created_at') .order('created_at', { ascending: false }); if (error) { console.error('Supabase request error:', error); box.append(el('p', { className: 'mut', textContent: 'خطا در دریافت درخواست‌ها: ' + error.message })); say('دریافت درخواست‌ها از Supabase انجام نشد.'); return; } requests = data || []; const pendingCount = requests.filter(r => r.status === 'pending').length; $('#reqCount').textContent = pendingCount ? pendingCount + ' درخواست جدید' : 'درخواست جدیدی نیست'; $('#reqTitle').textContent = `🌠 درخواست‌های عضویت (${requests.length})`; if (!requests.length) { box.append(el('p', { className: 'mut', textContent: 'هنوز هیچ درخواست عضویتی ثبت نشده است.' })); return; } const emails = new Set( members.map(m => (m.email || '').toLowerCase().trim()) ); requests.forEach(r => { const isMem = emails.has((r.email || '').toLowerCase().trim()); let statusText = 'در انتظار بررسی'; let statusClass = 'pill warn'; if (r.status === 'approved') { statusText = 'تأیید شده'; statusClass = 'pill good'; } else if (r.status === 'rejected') { statusText = 'رد شده'; statusClass = 'pill'; } const status = el('span', { className: statusClass, textContent: statusText }); const actions = el('span', { className: 'acts' }); if (r.status === 'pending') { if (isMem) { actions.append(el('span', { className: 'pill good', textContent: 'قبلاً به فهرست اعضا اضافه شده' })); } else { const add = el('button', { type: 'button', className: 'btn sm', textContent: '＋ افزودن به اعضا' }); add.onclick = async () => { if (!confirm( `درخواست ${r.name} تأیید و به فهرست اعضا اضافه شود؟` )) return; add.disabled = true; add.textContent = 'در حال انجام...'; // افزودن به فهرست اعضای فعلی const result = await call('madd', { people: [{ name: r.name, email: r.email }] }); if (!result.ok) { add.disabled = false; add.textContent = '＋ افزودن به اعضا'; fail(result); return; } // تغییر وضعیت درخواست در Supabase const { error: updateError } = await supabaseClient .from('membership_requests') .update({ status: 'approved' }) .eq('id', r.id); if (updateError) { console.error(updateError); say( 'عضو به فهرست اضافه شد، اما وضعیت درخواست تغییر نکرد: ' + updateError.message ); await loadReq(); return; } say('✨ درخواست ' + r.name + ' تأیید شد و به فهرست اعضا اضافه شد.'); await loadReq(); }; actions.append(add); } } const del = el('button', { type: 'button', className: 'del', textContent: 'حذف درخواست' }); del.onclick = async () => { if (!confirm(`درخواست ${r.name} حذف شود؟`)) return; const { error: deleteError } = await supabaseClient .from('membership_requests') .delete() .eq('id', r.id); if (deleteError) { console.error(deleteError); say('حذف درخواست انجام نشد: ' + deleteError.message); return; } say('درخواست حذف شد.'); await loadReq(); }; actions.append(del); let dateText = 'تاریخ نامشخص'; if (r.created_at) { dateText = 'تاریخ ثبت: ' + fdate(r.created_at); } box.append( el('div', { className: 'item' }, el('b', { textContent: r.name }), actions, el('p', { textContent: 'کلاس: ' + (r.cls || '—') + ' · شاد: ' + (r.shad || '—') + ' · ایمیل: ' + (r.email || '—'), dir: 'auto' }), el('small', {}, status, ' · ' + dateText) ) ); }); }
